vi docker-compose.yml
version: '3.8'

services:
  urbackup:
    image: uroni/urbackup-server:latest
    container_name: urbackup-server
    restart: unless-stopped
    environment:
      - PUID=1000
      - PGID=1000
    ports:
      - "55414:55414"
      - "55415:55415"
      - "35623:35623/udp"
    volumes:
      - /urbackup-storage:/backups
      - /urbackup-storage/urbackup_www:/var/urbackup
    network_mode: "host"

# 1. ディレクトリの所有者を PUID/PGID（1000:1000）に変更する
sudo chown -R 1000:1000 /urbackup-storage

# 2. 所有者に読み書き・実行権限を与える
sudo chmod -R 755 /urbackup-storage


sudo docker compose up -d
