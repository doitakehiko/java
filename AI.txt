 cat sample_ai.py
import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

# 1. GPU（RTX 2070）を使う設定
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"使用中のデバイス: {device}")

# 2. 画像データのダウンロードと前処理設定
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))
])

train_dataset = datasets.MNIST(root='./data', train=True, download=True, transform=transform)
train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)

# 3. シンプルなAIモデル（ニューラルネットワーク）の定義
class SimpleNet(nn.Module):
    def __init__(self):
        super(SimpleNet, self).__init__()
        self.fc1 = nn.Linear(28 * 28, 128)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(128, 10)

    def forward(self, x):
        x = x.view(-1, 28 * 28)
        x = self.relu(self.fc1(x))
        x = self.fc2(x)
        return x

model = SimpleNet().to(device)

# 4. 損失関数と最適化（学習のルール）の設定
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# 5. AI의 学習実行
model.train()
print("AIの学習を開始します。GPUのパワーを体感してください...")

for batch_idx, (data, target) in enumerate(train_loader):
    data, target = data.to(device), target.to(device)
    optimizer.zero_grad()
    output = model(data)
    loss = criterion(output, target)
    loss.backward()
    optimizer.step()

    if batch_idx % 100 == 0:
        print(f"進捗: [{batch_idx * len(data)}/{len(train_loader.dataset)}] 誤差(Loss): {loss.item():.6f}")

print("🎉 学習が正常に完了しました！RTX 2070での計算成功です。")

#end sample_ai.py

python3 sample_ai.py

import torch
from torchvision import datasets, transforms
from torch.utils.data import DataLoader
from sample_ai import SimpleNet # さっき作ったAIの形を再利用

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# 1. テスト専用の「別の1万枚の画像」を準備
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))
])
test_dataset = datasets.MNIST(root='./data', train=False, download=True, transform=transform)
test_loader = DataLoader(test_dataset, batch_size=1000, shuffle=False)

# 2. 先ほど学習させたAIモデルの状態をチェック
# ※今回は簡易的に、スクリプト内で再度一瞬だけ学習させてテストします
import sample_ai
model = sample_ai.model
model.eval() # テストモードに変える

correct = 0
total = 0

print("学習を終えたAIに、初めて見る1万枚のテスト問題を解かせています...")

with torch.no_grad(): # テストなので学習（逆伝播）はしない
    for data, target in test_loader:
        data, target = data.to(device), target.to(device)
        outputs = model(data)
        _, predicted = torch.max(outputs.data, 1)
        total += target.size(0)
        correct += (predicted == target).sum().item()

accuracy = 100 * correct / total
print(f"========= テスト結果 =========")
print(f"テスト画像：{total} 枚")
print(f"正解数　　：{correct} 枚")
print(f"正解率 (Accuracy): {accuracy:.2f} %")
print(f"==============================")

import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))
])

train_dataset = datasets.MNIST(root='./data', train=True, download=True, transform=transform)
train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)

class SimpleNet(nn.Module):
    def __init__(self):
        super(SimpleNet, self).__init__()
        self.fc1 = nn.Linear(28 * 28, 128)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(128, 10)

    def forward(self, x):
        x = x.view(-1, 28 * 28)
        x = self.relu(self.fc1(x))
        x = self.fc2(x)
        return x

model = SimpleNet().to(device)
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

model.train()
print("AIの再学習を開始します...")
for batch_idx, (data, target) in enumerate(train_loader):
    data, target = data.to(device), target.to(device)
    optimizer.zero_grad()
    output = model(data)
    loss = criterion(output, target)
    loss.backward()
    optimizer.step()

# ★ここが重要：学習した「重み（記憶）」をファイルに保存
torch.save(model.state_dict(), "mnist_model.pth")
print("🎉 AIの記憶を 'mnist_model.pth' として保存しました！")

 cat predict.py
import torch
import torch.nn as nn
from PIL import Image
from torchvision import transforms

#device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
device = torch.device("cpu")

# 1. AIの形を定義（学習時と同じ形にする）
class SimpleNet(nn.Module):
    def __init__(self):
        super(SimpleNet, self).__init__()
        self.fc1 = nn.Linear(28 * 28, 128)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(128, 10)

    def forward(self, x):
        x = x.view(-1, 28 * 28)
        x = self.relu(self.fc1(x))
        x = self.fc2(x)
        return x

# 2. AIを準備して保存した記憶（ファイルを）を読み込む
model = SimpleNet().to(device)
model.load_state_dict(torch.load("mnist_model.pth", map_location=device))
model.eval()

# 3. アップロードした手書き画像を読み込んでAI用に変換
image_path = "my_number.png"
try:
    img = Image.open(image_path).convert('L') # グレースケール（白黒）に変換
except FileNotFoundError:
    print(f"エラー: {image_path} が見つかりません。画像をアップロードしてくださ い。")
    exit()

# AIが学習した時と同じ規格（28x28、テンソル化、正規化）に変換
transform = transforms.Compose([
    transforms.Resize((28, 28)),
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))
])
img_tensor = transform(img).unsqueeze(0).to(device) # バッチの次元を追加してGPU へ

# 4. AIの判定
with torch.no_grad():
    output = model(img_tensor)
    _, predicted = torch.max(output, 1)

print(f"========= 判定結果 =========")
print(f"AIの予想した数字は: 【 {predicted.item()} 】 です！")
print(f"============================")
